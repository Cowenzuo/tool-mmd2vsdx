  <DocumentSettings TopPage="0"
     DefaultTextStyle="3" DefaultLineStyle="3"

    <GlueSettings>9</GlueSettings>
    <SnapSettings>295</SnapSettings>
    <DynamicGridEnabled>1</DynamicGridEnabled>
    <ProtectStyles>0</ProtectStyles>
    ...
  </DocumentSettings>

  <Colors>
    <ColorEntry IX="0" RGB="#000000"/>
  </Colors>

  <FaceNames>

  </FaceNames>

  <StyleSheets>
    <StyleSheet ID="0" NameU="No Style">...</StyleSheet>
    <StyleSheet ID="3" NameU="Normal">...</StyleSheet>
