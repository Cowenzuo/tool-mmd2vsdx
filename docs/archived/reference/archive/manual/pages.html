5. pages.xml & pageN.xml

pages.xml — 页面索引

<Pages xmlns="...visio/2012/main"
       xmlns:r="...relationships">
  <Page ID="0" NameU="Page-1" Name="Page-1"
        ViewScale="1" ViewCenterX="4.25" ViewCenterY="5.5">
    <PageSheet LineStyle="0" FillStyle="0" TextStyle="0">
      <Cell N="PageWidth" V="8.5" U="IN"/>
      <Cell N="PageHeight" V="11" U="IN"/>
      <Cell N="PageScale" V="1" U="IN"/>
      <Cell N="DrawingScale" V="1" U="IN"/>
      <Cell N="DrawingSizeType" V="0"/>
      <Cell N="DrawingScaleType" V="0"/>
      <Cell N="InhibitSnap" V="0"/>
      <Cell N="UIVisibility" V="0"/>
      <Cell N="DrawingResizeType" V="1"/>
      <Cell N="PlaceStyle" V="2"/>
      <Cell N="RouteStyle" V="6"/>
      <Cell N="PageShapeSplit" V="1"/>
    </PageSheet>
    <Rel r:id="rId1"/>  <!-- 指向 page1.xml -->
  </Page>
</Pages>

pageN.xml — 页面内容

<PageContents xmlns="...visio/2012/main"
              xmlns:r="...relationships">
  <Shapes>
    <Shape ID="1" ...>...</Shape>
    ...
  </Shapes>
  <Connects>
    <Connect FromSheet="5" FromCell="BeginX" FromPart="9"
             ToSheet="1" ToCell="Connections.X2" ToPart="101"/>
    ...
  </Connects>
</PageContents>

页面尺寸单位

PageWidth/PageHeight 使用英寸 (U="IN")。坐标系统中 Y 轴向下增加，需要做 Y 翻转转换。
