2. [Content_Types].xml

<Types xmlns="...content-types">
  <Default Extension="rels"
    ContentType="application/vnd.openxmlformats-package.relationships+xml"/>
  <Default Extension="xml" ContentType="application/xml"/>
  <Override PartName="/visio/document.xml"
    ContentType="application/vnd.ms-visio.drawing.main+xml"/>
  <Override PartName="/visio/pages/pages.xml"
    ContentType="application/vnd.ms-visio.pages+xml"/>
  <Override PartName="/visio/pages/page1.xml"
    ContentType="application/vnd.ms-visio.page+xml"/>
  <Override PartName="/visio/masters/masters.xml"
    ContentType="application/vnd.ms-visio.masters+xml"/>

    ContentType="application/vnd.ms-visio.master+xml"/>

3. Relationships (.rels)

每个部件可有关联的 .rels 文件，存储在同一目录的 _rels/ 子目录中。

包级别 (_rels/.rels)

<Relationships xmlns="...relationships">
  <Relationship Id="rId1" Type="...relationships/document"
    Target="visio/document.xml"/>
  <Relationship Id="rId2" Type="...metadata/core-properties"
    Target="docProps/core.xml"/>
</Relationships>

文档级别 (visio/_rels/document.xml.rels)

<Relationship Id="rId1" Type="...relationships/masters"
  Target="masters/masters.xml"/>
<Relationship Id="rId2" Type="...relationships/pages"
  Target="pages/pages.xml"/>
<Relationship Id="rId3" Type="...relationships/windows"
  Target="windows.xml"/>
